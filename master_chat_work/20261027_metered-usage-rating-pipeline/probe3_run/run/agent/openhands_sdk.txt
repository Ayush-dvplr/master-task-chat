    def allocate(lines, total_cents, pool_cents):
        """Share pool_cents over lines in proportion to charge, per the 
contract."""
        result = [0] * len(lines)
        if not lines or total_cents <= 0 or pool_cents <= 0:
            return result
        first = [round_half_even(Fraction(pool_cents * charge, total_cents)) for
charge in lines]
        kept = [index for index, value in enumerate(first) if value >= 50]
        if not kept:
            return result
        kept_total = sum(lines[index] for index in kept)
        if kept_total <= 0:
            return result
        for index in kept:
            result[index] = round_half_even(Fraction(pool_cents * lines[index], 
kept_total))
        return result
