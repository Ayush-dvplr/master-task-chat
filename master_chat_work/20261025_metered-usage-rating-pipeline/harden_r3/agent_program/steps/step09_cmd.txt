s=open('collector_south.txt',encoding='latin-1').read().split('\n')
print('south lens', collections.Counter(len(l) for l in s if l).most_common(10), len(s))
print('south units bad', collections.Counter(l[46:52] for l in s if l and not l[46:52].strip().isdigit()))
# west
w=open('collector_west.log',encoding='utf-8').read().split('\n')
print('west count', len([l for l in w if l]))
print('west sample bad qty', collections.Counter(re.search(r'qty=(\S*)',l).group(1) for l in w if l and re.search(r'qty=(\S*)',l) and not re.search(r'qty=(\S*)',l).group(1).isdigit()))
print([l for l in w if l and 'qty=' not in l][:5])
